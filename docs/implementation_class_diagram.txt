class EKMoveValidator

' ================= 3. ЛОГИКА: РЕЗУЛЬТАТ ХОДА =================
' ИСПРАВЛЕНО: AwaitingNope возвращён — окно Nope является частью правил
' игры, а не деталью реализации, которую можно было опустить.
' AwaitingNope отличается от AwaitingResponse: ответить Nope может
' ЛЮБОЙ игрок (нет конкретного responderId), а на Favor/тройку отвечает
' только один конкретный игрок.
abstract class ValidationResult
class Accepted {
  +move: Move
}
class Rejected {
  +errors: List<String>
}
class AwaitingNope {
  +pendingMove: Move
}
class AwaitingResponse {
  +pendingMove: Move
  +responderId: Int
}
ValidationResult <|-- Accepted
ValidationResult <|-- Rejected
ValidationResult <|-- AwaitingNope
ValidationResult <|-- AwaitingResponse

' ================= 4. ЛОГИКА: СЕРВИСЫ =================
class GameplayService {
  -historyRepo: IHistoryRepository
  -validator: IMoveValidator
  -playerRegistry: PlayerRegistryService
  -statistics: StatisticsService
  +startGame(playerNames: List<String>): Int
  +playMove(move: Move): ValidationResult
  +resolveNopeWindow(gameId: Int): ValidationResult
  +endGame(gameId: Int): void
  +getCurrentGame(gameId: Int): Game
}
GameplayService --> IHistoryRepository : сохраняет партии и ходы
GameplayService --> IMoveValidator : проверяет правила ходов
GameplayService --> PlayerRegistryService : регистрирует игроков партии
GameplayService --> StatisticsService : обновляет статистику по завершении
GameplayService ..> ValidationResult : возвращает результат хода

class PlayerRegistryService {
  -repo: IPlayerRepository
  +getKnownPlayerNames(): List<String>
  +registerPlayer(name: String): Int
  +findPlayerId(name: String): Int?
}
PlayerRegistryService --> IPlayerRepository

class HistoryService {
  -repo: IHistoryRepository
  +getHistory(gameId: Int): List<Move>
  +getAllFinishedGames(): List<GameSummary>
}
HistoryService --> IHistoryRepository : читает историю
HistoryService ..> GameSummary : формирует сводку партии

class StatisticsService {
  -repo: IStatisticsRepository
  +getLeaderboard(): List<PlayerStats>
  +getPlayerStats(playerId: Int): PlayerStats
  +getAllPlayerStats(): List<PlayerStats>
  +updateStats(gameSummary: GameSummary): void
  -mergeStats(current: PlayerStats, gameSummary: GameSummary): PlayerStats
}
StatisticsService --> IStatisticsRepository : читает текущий PlayerStats,\nпишет пересчитанный PlayerStats
StatisticsService ..> PlayerStats
