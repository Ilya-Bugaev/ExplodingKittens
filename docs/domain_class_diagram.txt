@startuml
skinparam classAttributeIconSize 0
skinparam linetype ortho
title Взрывные котята — Предметная Диаграмма Классов

' ================= ПЕРЕЧИСЛЕНИЯ =================
enum CardType {
    EXPLODING_KITTEN
    DEFUSE
    NOPE
    ATTACK
    SKIP
    FAVOR
    SHUFFLE
    SEE_FUTURE
    CAT_BEARD
    CAT_TACO
    CAT_HAIRY_POTATO
    CAT_CATERMELON
    CAT_RAINBOW_RALPHING
    --
    +isCatCard: Boolean
    +displayName: String
}

enum MoveType {
    START
    DRAW
    PLAY_CARD
    PLAY_TWO_OF_A_KIND
    PLAY_THREE_OF_A_KIND
    PLAY_FIVE_DIFFERENT
    RESOLVE_PENDING
}

enum GameState {
    SETUP
    IN_PROGRESS
    FINISHED
}

' ================= КАРТА =================
class Card {
    +id: Int
    +type: CardType
    +isExplodingKitten: Boolean
}

' ================= ИГРОК =================
class Player {
    +id: Int
    +name: String
    +isAlive: Boolean
    +addCard(card: Card): Unit
    +removeCard(card: Card): Boolean
    +hasCardOfType(type: CardType): Boolean
    +findCardsOfType(type: CardType): List<Card>
    +eliminate(): Unit
}

' ================= КОЛОДА И СБРОС =================
class Deck {
    +shuffle(random: Random = Random.Default): Unit
    +draw(): Card
    +peekTop(amount: Int): List<Card>
    +insertCardAt(position: Int, card: Card): Unit
    +insertCardAtTop(card: Card): Unit
    +size(): Int
    +isEmpty(): Boolean
}

class DiscardPile {
    +add(card: Card): Unit
    +addAll(cards: List<Card>): Unit
    +takeAnyCard(): Card?
    +takeCardOfType(type: CardType): Card?
    +size(): Int
    +isEmpty(): Boolean
}

' ================= ХОД =================
class Move {
    +id: Int
    +turnNumber: Int
    +type: MoveType
    +cardsPlayed: List<Card>
    +drawnCard: Card?
    +receivedCard: Card?
    +requestedCardType: CardType?
    +placedKittenPosition: Int?
    +eliminated: Boolean
    +initialHands: Map<Int, List<Card>>?
    +initialDeckOrder: List<Card>?
    +endsTurn: Boolean
}

' ================= ПАРТИЯ =================
class Game {
    +id: Int
    +state: GameState
    +currentTurnIndex: Int
    +turnsPlayed: Int
    +attacksPending: Int
    +startGame(playerNames: List<String>, random: Random = Random.Default): Unit
    +getCurrentPlayer(): Player
    +getAlivePlayers(): List<Player>
    +isFinished(): Boolean
    +finish(): Unit
    +addMove(move: Move): Move
    +advanceTurn(forAttack: Boolean): Unit
    +setAttacksPending(value: Int): Unit
    +setPendingKitten(card: Card): Unit
    +clearPendingKitten(): Unit
    +setPendingMove(move: Move): Unit
    +clearPendingMove(): Unit
    +getInitialSetup(): String
}

' ================= СВЯЗИ =================
Game "1" *-- "2..5" Player : участники
Game "1" *-- "1" Deck : колода добора
Game "1" *-- "1" DiscardPile : сброс
Game "1" *-- "0..*" Move : история ходов

Player "1" o-- "0..*" Card : рука
Deck "1" o-- "0..*" Card : карты в колоде
DiscardPile "1" o-- "0..*" Card : сброшенные карты

Move "0..*" --> "0..1" Player : автор
Move "0..*" --> "0..1" Player : цель

Move "0..1" --> "0..1" Move : отменяет

Game "1" --> "0..1" Player : победитель
Game "1" --> "0..1" Move : ожидает разрешения (окно Nope / ответ игрока)
Game "1" --> "0..1" Card : ожидает обезвреживания

@enduml
